- Function(s): sqlite3PutVarint (37991), and putVarint64 (37967) when called directly.
- Clause, as CBMC states it, identical in both contracts:

  ```c
  __CPROVER_requires(__CPROVER_w_ok(p, (1 + (v >= ((u64)1 << 7)) + (v >= ((u64)1 << 14))
      + (v >= ((u64)1 << 21)) + (v >= ((u64)1 << 28))
      + (v >= ((u64)1 << 35)) + (v >= ((u64)1 << 42))
      + (v >= ((u64)1 << 49)) + (v >= ((u64)1 << 56)))))

Initial bytes are unrestricted; the value only determines the required storage size.

from pathlib import Path
import os
root = Path(os.environ['TMPDIR']) / 'C-01-b4'
p = root / 'matrix' / 'C-01'
p.mkdir(parents=True, exist_ok=True)
h = ['#include "sqlite3.c"\n']
mk, names = ['CARD := C-01'], []
for k in range(1, 10):
    lo = 0 if k == 1 else 1 << (7*(k-1))
    hi = (1 << (7*k))-1 if k < 9 else (1 << 64)-1
    for short in (False, True):
        for fn, name in [('putVarint64', 'put64'), ('sqlite3PutVarint', 'put')]:
            n = f'{name}_{k}_{"short" if short else "exact"}'
            names.append(n)
            h.append(f'''void harness_{n}(void){{
  unsigned char *output;
  u64 value;
  /* SETUP */
  __CPROVER_assume(value >= {lo}ULL && value <= {hi}ULL);
  output = malloc({k-int(short)});
  __CPROVER_assume(output != 0);
  /* END SETUP */
  {fn}(output, value);
  free(output);
}}
''')
            mk += [f'{n}_ENTRY := harness_{n}', f'{n}_LOOPS := unwind',

mk += ['include ../Makefile.common']

(p/'Makefile').write_text('\n'.join(mk)+'\n')
